# V0.6e Feedback Streaming + Prewriting Reliability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship V0.6e structured essay-feedback streaming, prewriting progress jobs, reliable material-card generation visibility, and accurate backend-owned usage/cost accounting.

**Architecture:** Add a narrow streaming layer around the existing Prompt Registry, ModelRouter, AI runner, essay routes, and Writing Castle routes. Store official learning state only after schema validation; use `EssayFeedbackSubmission` for idempotency and daily-limit finalization; use `PrewritingAIJob` for material-card and outline progress without streaming partial JSON.

**Tech Stack:** FastAPI, SQLModel, Alembic, pytest, Pydantic, httpx, Next.js App Router, React, Vitest, Testing Library, Playwright, browser `fetch` streams for POST SSE, and native `EventSource` for GET progress jobs.

## Scope Guardrails

Implement the five V0.6e P0 items together because daily limits, logging,
streaming UX, and material-card reliability all meet at the same AI execution
boundary:

- `essay_feedback` structured card streaming for direct draft and Writing Castle first draft;
- `LLMCallLog` streaming fields plus authoritative usage/cost accounting;
- daily-limit reservation rules for streaming, fallback, and disconnect;
- Progress SSE jobs for material cards and outlines;
- material-card reliability metrics and fixed-case QA gate.

Do not implement sentence streaming, revision-comparison streaming, a generic
job platform, a new Admin dashboard, or a provider benchmark framework.

Create `apps/api/app/services/llm_usage_accounting.py`:

```python
from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class UsageCostRecord:
    usage_available: bool
    usage_source: str
    usage_is_estimated: bool
    usage_details_json: dict[str, Any]
    prompt_tokens: int | None
    completion_tokens: int | None
    total_tokens: int | None
    estimated_cost_usd: float | None
    cost_source: str
    pricing_snapshot_id: str | None
    pricing_snapshot_version: str
    provider_reported_cost_usd: float | None
    cost_calculation_version: str


def _int_or_none(value: Any) -> int | None:
    if value is None:
        return None
    return int(value)


def normalize_usage_and_cost(
    *,
    provider: str,
    model: str,
    role: str,
    provider_usage: dict[str, Any] | None,
    usage_source: str,
    tokenizer_estimate: dict[str, int] | None,
    provider_reported_cost_usd: float | None,
    input_cost_per_1k: float,
    output_cost_per_1k: float,
    delegate_to_provider_reported_cost: bool = False,
) -> UsageCostRecord:
    snapshot_version = "v0.6e.1"
    snapshot_id = f"settings:{provider}:{model}:{role}:{snapshot_version}"

    if provider_usage is not None and usage_source != "tokenizer_estimate":
        prompt_tokens = _int_or_none(provider_usage.get("prompt_tokens"))
        completion_tokens = _int_or_none(provider_usage.get("completion_tokens"))
        total_tokens = _int_or_none(provider_usage.get("total_tokens"))
        cost = None
        cost_source = "unavailable"
        if delegate_to_provider_reported_cost and provider_reported_cost_usd is not None:
            cost = round(float(provider_reported_cost_usd), 8)
            cost_source = "provider_reported_cost"
        elif prompt_tokens is not None and completion_tokens is not None:
            cost = round(
                (prompt_tokens / 1000 * input_cost_per_1k)
                + (completion_tokens / 1000 * output_cost_per_1k),
                8,
            )
            cost_source = "provider_usage_x_pricing_snapshot"
        return UsageCostRecord(
            usage_available=True,
            usage_source=usage_source,
            usage_is_estimated=False,
            usage_details_json={
                "prompt_tokens": prompt_tokens,
                "completion_tokens": completion_tokens,
                "total_tokens": total_tokens,
                "cached_input_tokens": provider_usage.get("cached_input_tokens"),
                "reasoning_tokens": provider_usage.get("reasoning_tokens"),
                "thoughts_tokens": provider_usage.get("thoughts_tokens"),
                "audio_tokens": provider_usage.get("audio_tokens"),
                "image_tokens": provider_usage.get("image_tokens"),
                "provider_raw_usage": provider_usage,
            },
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            total_tokens=total_tokens,
            estimated_cost_usd=cost,
            cost_source=cost_source,
            pricing_snapshot_id=snapshot_id,
            pricing_snapshot_version=snapshot_version,
            provider_reported_cost_usd=provider_reported_cost_usd,
            cost_calculation_version=snapshot_version,
        )

    details = {
        "prompt_tokens": None,
        "completion_tokens": None,
        "total_tokens": None,
        "cached_input_tokens": None,
        "reasoning_tokens": None,
        "thoughts_tokens": None,
        "audio_tokens": None,
        "image_tokens": None,
        "provider_raw_usage": None,
        "usage_unavailable_reason": "provider_usage_missing",
    }
    if tokenizer_estimate is not None:
        details.update(
            {
                "estimated_prompt_tokens": tokenizer_estimate.get("prompt_tokens"),
                "estimated_completion_tokens": tokenizer_estimate.get("completion_tokens"),
                "estimated_total_tokens": tokenizer_estimate.get("total_tokens"),
            }
        )

    return UsageCostRecord(
        usage_available=False,
        usage_source=usage_source,
        usage_is_estimated=usage_source == "tokenizer_estimate",
        usage_details_json=details,
        prompt_tokens=None,
        completion_tokens=None,
        total_tokens=None,
        estimated_cost_usd=None,
        cost_source="unavailable",
        pricing_snapshot_id=None,
        pricing_snapshot_version=snapshot_version,
        provider_reported_cost_usd=provider_reported_cost_usd,
        cost_calculation_version=snapshot_version,
    )
```

```python
from hashlib import sha256
import json
from typing import Any

from sqlmodel import Session, select

from app.domain.models import EssayFeedbackSubmission, new_uuid, utcnow


class IdempotencyPayloadMismatch(Exception):
    """Raised when the same client_submission_id is reused with different content."""


TRANSPORT_FIELDS = {"client_submission_id", "timestamp", "auth", "session"}


def _canonicalize_value(value: Any) -> Any:
    if value == "":
        return None
    if isinstance(value, dict):
        return {
            key: _canonicalize_value(value[key])
            for key in sorted(value)
            if key not in TRANSPORT_FIELDS
        }
    if isinstance(value, list):
        return [_canonicalize_value(item) for item in value]
    return value


def build_submission_payload_hash(
    *,
    task_name: str,
    route_scope: str,
    payload_schema_version: str,
    payload: dict[str, Any],
) -> str:
    canonical = {
        "payload_schema_version": payload_schema_version,
        "route_scope": route_scope,
        "task_name": task_name,
        "payload": _canonicalize_value(payload),
    }
    encoded = json.dumps(canonical, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    return sha256(encoded.encode("utf-8")).hexdigest()
```

In `apps/api/app/services/essay_feedback_streaming.py`, define:

```python
from dataclasses import dataclass
import re


SECTION_ORDER = ["strengths", "improvements", "sentence_notes", "revision_tasks"]
SECTION_PATTERN = re.compile(r"<(?P<name>strengths|improvements|sentence_notes|revision_tasks)>"
                             r"(?P<body>.*?)"
                             r"</(?P=name)>", re.DOTALL)


@dataclass(frozen=True)
class FeedbackSectionPreview:
    section: str
    items: list[str]


class StreamSectionError(Exception):
    def __init__(self, code: str, message: str):
        super().__init__(message)
        self.code = code


class EssayFeedbackSectionParser:
    def __init__(self, *, max_buffer_bytes: int = 12000):
        self.buffer = ""
        self.next_index = 0
        self.seen: set[str] = set()
        self.max_buffer_bytes = max_buffer_bytes

    def feed(self, text_delta: str) -> list[FeedbackSectionPreview]:
        self.buffer += text_delta
        if len(self.buffer.encode("utf-8")) > self.max_buffer_bytes:
            raise StreamSectionError("STREAM_SECTION_TOO_LARGE", "section buffer exceeded limit")

        emitted: list[FeedbackSectionPreview] = []
        while True:
            match = SECTION_PATTERN.search(self.buffer)
            if match is None:
                return emitted
            name = match.group("name")
            if name in self.seen:
                raise StreamSectionError("STREAM_SECTION_DUPLICATE", name)
            expected = SECTION_ORDER[self.next_index]
            if name != expected:
                raise StreamSectionError("STREAM_SECTION_OUT_OF_ORDER", name)
            items = [
                line.strip()[2:].strip()
                for line in match.group("body").splitlines()
                if line.strip().startswith("- ")
            ]
            if not items:
                raise StreamSectionError("STREAM_FINAL_SCHEMA_INVALID", name)
            self.seen.add(name)
            self.next_index += 1
            emitted.append(FeedbackSectionPreview(section=name, items=items))
            self.buffer = self.buffer[match.end():]
```

```typescript
export type SseFrame = {
  event: string;
  data: Record<string, unknown>;
};

export function parseSseFrames(chunk: string): SseFrame[] {
  return chunk
    .split("\n\n")
    .filter((frame) => frame.trim().length > 0)
    .map((frame) => {
      const eventLine = frame.split("\n").find((line) => line.startsWith("event: "));
      const dataLine = frame.split("\n").find((line) => line.startsWith("data: "));
      return {
        event: eventLine ? eventLine.slice("event: ".length) : "message",
        data: dataLine ? JSON.parse(dataLine.slice("data: ".length)) : {},
      };
    });
}

export type StreamReducerState = {
  lastSeq: number;
  sections: Record<string, string[]>;
  done: boolean;
  fetchUrl: string | null;
};

export function reduceStreamEvent(
  state: StreamReducerState | undefined,
  frame: SseFrame,
): StreamReducerState {
  const current = state ?? { lastSeq: 0, sections: {}, done: false, fetchUrl: null };
  const seq = Number(frame.data.seq ?? 0);
  if (seq <= current.lastSeq) {
    return current;
  }
  if (frame.event === "feedback_section_preview") {
    const section = String(frame.data.section);
    return {
      ...current,
      lastSeq: seq,
      sections: { ...current.sections, [section]: frame.data.items as string[] },
    };
  }
  if (frame.event === "done") {
    const result = frame.data.result as { fetch_url?: string } | undefined;
    return { ...current, lastSeq: seq, done: true, fetchUrl: result?.fetch_url ?? null };
  }
  return { ...current, lastSeq: seq };
}
```

```typescript
export async function streamEssayFeedback(
  studentId: string,
  payload: EssayFeedbackStreamPayload,
  onFrame: (frame: EssayFeedbackStreamFrame) => void,
): Promise<void> {
  const response = await fetch(apiUrl(`/api/students/${studentId}/essays/stream-feedback`), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
  });
  if (!response.ok || !response.body) {
    throw new Error("stream request failed");
  }
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const parts = buffer.split("\n\n");
    buffer = parts.pop() ?? "";
    for (const part of parts) {
      for (const frame of parseSseFrames(`${part}\n\n`)) {
        onFrame(frame);
      }
    }
  }
}
```
