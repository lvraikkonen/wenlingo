# V0.6e Feedback Streaming + Prewriting Reliability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship V0.6e structured essay-feedback streaming, prewriting progress jobs, reliable material-card generation visibility, and accurate backend-owned usage/cost accounting.

**Architecture:** Add a narrow streaming layer around the existing Prompt Registry, ModelRouter, AI runner, essay routes, and Writing Castle routes. Store official learning state only after schema validation; use `EssayFeedbackSubmission` for idempotency and daily-limit finalization; use `PrewritingAIJob` for material-card and outline progress without streaming partial JSON.

**Tech Stack:** FastAPI, SQLModel, Alembic, pytest, Pydantic, httpx, Next.js App Router, React, Vitest, Testing Library, Playwright, browser `fetch` streams for POST SSE, and native `EventSource` for GET progress jobs.

## Scope Guardrails

Implement the five V0.6e P0 items together because daily limits, logging,
streaming UX, and material-card reliability all meet at the same AI execution
boundary:

- `essay_feedback` structured card streaming for direct draft and Writing Castle first draft;
- `LLMCallLog` streaming fields plus authoritative usage/cost accounting;
- daily-limit reservation rules for streaming, fallback, and disconnect;
- Progress SSE jobs for material cards and outlines;
- material-card reliability metrics and fixed-case QA gate.

Do not implement sentence streaming, revision-comparison streaming, a generic
job platform, a new Admin dashboard, or a provider benchmark framework.

def calculate_cost(
    *,
    usage: NormalizedUsage,
    pricing: ModelPricing | None,
    provider_reported_cost_usd: float | None,
):
    if pricing is None:
        return None, "unavailable", "PRICING_UNCONFIGURED"
    if not usage.usage_available:
        return None, "unavailable", ""
    if pricing.delegates_to_provider_reported_cost:
        if provider_reported_cost_usd is None:
            return None, "unavailable", "PROVIDER_REPORTED_COST_MISSING"
        return round(float(provider_reported_cost_usd), 8), "provider_reported_cost", ""

    details = usage.usage_details_json
    if details.get("reasoning_tokens") and not pricing.reasoning_cost_per_1k_tokens:
        return None, "unavailable", "UNSUPPORTED_USAGE_CLASS"
    if details.get("thoughts_tokens") and not pricing.thoughts_cost_per_1k_tokens:
        return None, "unavailable", "UNSUPPORTED_USAGE_CLASS"
    if details.get("cached_input_tokens") and not pricing.cached_input_cost_per_1k_tokens:
        return None, "unavailable", "UNSUPPORTED_USAGE_CLASS"
    if not pricing.input_cost_per_1k_tokens or not pricing.output_cost_per_1k_tokens:
        return None, "unavailable", "PRICING_UNCONFIGURED"

    cached_input_tokens = int(details.get("cached_input_tokens") or 0)
    cached_in_prompt = details.get("cached_input_tokens_included_in_prompt_tokens") is not False
    prompt_tokens = usage.prompt_tokens or 0
    uncached_prompt_tokens = (
        max(prompt_tokens - cached_input_tokens, 0) if cached_in_prompt else prompt_tokens
    )
    cost = uncached_prompt_tokens / 1000 * pricing.input_cost_per_1k_tokens
    cost += (usage.completion_tokens or 0) / 1000 * pricing.output_cost_per_1k_tokens
    cost += cached_input_tokens / 1000 * (
        pricing.cached_input_cost_per_1k_tokens or 0
    )
    cost += (details.get("reasoning_tokens") or 0) / 1000 * (
        pricing.reasoning_cost_per_1k_tokens or 0
    )
    cost += (details.get("thoughts_tokens") or 0) / 1000 * (
        pricing.thoughts_cost_per_1k_tokens or 0
    )
    return round(cost, 8), "provider_usage_x_pricing_snapshot", ""


def normalize_usage_and_cost(
    *,
    provider: str,
    model: str,
    role: str,
    provider_usage: dict[str, Any] | None,
    usage_source: str,
    tokenizer_estimate: dict[str, int] | None,
    provider_reported_cost_usd: float | None,
    pricing: ModelPricing | None,
) -> UsageCostRecord:

        provider_usage=provider_usage,
        usage_source=usage_source,
        tokenizer_estimate=tokenizer_estimate,
    )
    cost, cost_source, cost_error_code = calculate_cost(
        usage=usage,
        pricing=pricing,

    )

        usage_available=usage.usage_available,

from hashlib import sha256
import json
from typing import Any

from sqlmodel import Session, select

from app.domain.models import EssayFeedbackSubmission


class IdempotencyPayloadMismatch(Exception):
    """Raised when the same client_submission_id is reused with different content."""


TRANSPORT_FIELDS = {"client_submission_id", "timestamp", "auth", "session"}


def _canonicalize_value(value: Any) -> Any:
    if value == "":
        return None
    if isinstance(value, dict):
        return {
            key: _canonicalize_value(value[key])
            for key in sorted(value)
            if key not in TRANSPORT_FIELDS
        }
    if isinstance(value, list):
        return [_canonicalize_value(item) for item in value]
    return value


def build_submission_payload_hash(
    *,
    task_name: str,
    route_scope: str,
    payload_schema_version: str,
    payload: dict[str, Any],
) -> str:
    canonical = {
        "payload_schema_version": payload_schema_version,
        "route_scope": route_scope,
        "task_name": task_name,
        "payload": _canonicalize_value(payload),
    }
    encoded = json.dumps(canonical, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    return sha256(encoded.encode("utf-8")).hexdigest()

```python
from dataclasses import dataclass
import re


SECTION_ORDER = [
    "strengths", "improvements", "problem_monsters", "sentence_notes", "revision_tasks"
]
SECTION_PATTERN = re.compile(
    r"<(?P<name>strengths|improvements|problem_monsters|sentence_notes|revision_tasks)>"
    r"(?P<body>.*?)"
    r"</(?P=name)>",
    re.DOTALL,
)
SECTION_ITEM_LIMITS: dict[str, tuple[int, int]] = {
    "strengths": (2, 2),
    "improvements": (1, 3),
    "problem_monsters": (1, 3),
    "sentence_notes": (1, 3),
    "revision_tasks": (1, 1),
}
# problem_monsters is parsed into canonical EssayFeedback only, not emitted as a preview card.
# Per spec 9.1, preview sections are strengths/improvements/sentence_notes/revision_tasks.
PREVIEW_SECTIONS = {"strengths", "improvements", "sentence_notes", "revision_tasks"}


@dataclass(frozen=True)
class FeedbackSectionPreview:
    section: str
    items: list[str]


class StreamSectionError(Exception):
    def __init__(self, code: str, message: str):
        super().__init__(message)
        self.code = code


class EssayFeedbackSectionParser:
    def __init__(self, *, max_buffer_bytes: int = 12000, max_item_chars: int = 160):
        self.buffer = ""
        self.next_index = 0
        self.seen: set[str] = set()
        self.max_buffer_bytes = max_buffer_bytes
        self.max_item_chars = max_item_chars

    def feed(self, text_delta: str) -> list[FeedbackSectionPreview]:
        self.buffer += text_delta
        if len(self.buffer.encode("utf-8")) > self.max_buffer_bytes:
            raise StreamSectionError("STREAM_SECTION_TOO_LARGE", "section buffer exceeded limit")
        if "```" in self.buffer:
            raise StreamSectionError("STREAM_FINAL_SCHEMA_INVALID", "markdown code fence")

        emitted: list[FeedbackSectionPreview] = []
        while True:
            stripped = self.buffer.lstrip()
            if not stripped:
                self.buffer = ""
                return emitted
            expected = SECTION_ORDER[self.next_index]
            expected_open = f"<{expected}>"
            if not stripped.startswith(expected_open):
                if "<" in stripped or "\n" in stripped:
                    raise StreamSectionError("STREAM_SECTION_OUT_OF_ORDER", stripped[:40])
                return emitted
            match = SECTION_PATTERN.match(stripped)
            if match is None:
                return emitted
            name = match.group("name")
            if name in self.seen:
                raise StreamSectionError("STREAM_SECTION_DUPLICATE", name)
            if name != expected:
                raise StreamSectionError("STREAM_SECTION_OUT_OF_ORDER", name)
            body = match.group("body")
            if any(f"<{section}>" in body for section in SECTION_ORDER):
                raise StreamSectionError("STREAM_FINAL_SCHEMA_INVALID", "nested section")
            items = [
                line.strip()[2:].strip()
                for line in body.splitlines()
                if line.strip().startswith("- ")
            ]
            min_items, max_items = SECTION_ITEM_LIMITS.get(name, (1, 3))
            if not items or len(items) < min_items or len(items) > max_items or any(
                len(item) > self.max_item_chars for item in items
            ):
                raise StreamSectionError("STREAM_FINAL_SCHEMA_INVALID", name)
            if any("范文" in item or "可以这样写：" in item for item in items):
                raise StreamSectionError("STREAM_ANTI_GHOSTWRITING_BLOCKED", name)
            if name == "revision_tasks":
                for item in items:
                    parts = item.split(" | ", 1)
                    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                        raise StreamSectionError(
                            "STREAM_FINAL_SCHEMA_INVALID",
                            f"revision_task must be 'instruction | target', got: {item[:60]}",
                        )
            self.seen.add(name)
            self.next_index += 1
            if name in PREVIEW_SECTIONS:
                emitted.append(FeedbackSectionPreview(section=name, items=items))
            self.buffer = stripped[match.end():]
```

class StreamingBackendTask:
    async def run_to_terminal(self) -> None:
        await self.queue.put(self.event_builder.frame("start", {"phase": "reserved"}))
        async for provider_event in self.provider.stream_text("essay_feedback", self.payload):
            if provider_event.event_type == "text_delta":
                await self.handle_text_delta(provider_event.text_delta)
            elif provider_event.event_type == "usage_final":
                self.usage_record = normalize_usage_and_cost(
                    provider=self.provider.provider_name,
                    model=self.provider.model_name,
                    role="primary",
                    provider_usage=provider_event.usage,
                    usage_source="provider_final_chunk" if provider_event.usage else "unavailable",
                    tokenizer_estimate=None,
                    provider_reported_cost_usd=provider_event.provider_reported_cost_usd,
                    pricing=self.pricing,
                )
            elif provider_event.event_type == "provider_error":
                await self.fail_stream(provider_event.error_code)
                return
            elif provider_event.event_type == "provider_done":
                break
        await self.validate_save_and_emit_done()

    async def handle_text_delta(self, text_delta: str) -> None:
        previews = self.parser.feed(text_delta)
        for preview in previews:
            if self.first_visible_content_at is None:
                self.mark_first_visible_content()
            await self.queue.put(
                self.event_builder.frame(
                    "feedback_section_preview",
                    {"section": preview.section, "items": preview.items},
                )
            )


class SseMirror:
    async def frames(self):
        self.backend_task_handle = asyncio.create_task(self.backend_task.run_to_terminal())
        shielded = asyncio.shield(self.backend_task_handle)
        while True:
            frame = await self.backend_task.next_event_or_heartbeat()
            yield format_sse_event(frame.event_name, frame.data)
            if frame.event_name in {"done", "error"}:
                await shielded
                return

```typescript
export type SseFrame = {
  event: string;
  data: Record<string, unknown>;
};

export function parseSseFrames(chunk: string): SseFrame[] {
  return chunk
    .replaceAll("\r\n", "\n")
    .split("\n\n")
    .filter((frame) => frame.trim().length > 0)
    .filter((frame) => !frame.trimStart().startsWith(":"))
    .flatMap((frame) => {
      const eventLine = frame.split("\n").find((line) => line.startsWith("event: "));
      const dataLine = frame.split("\n").find((line) => line.startsWith("data: "));
      if (!dataLine) {
        return [];
      }
      return [{
        event: eventLine ? eventLine.slice("event: ".length) : "message",
        data: JSON.parse(dataLine.slice("data: ".length)),
      }];
    });
}

export type StreamReducerState = {
  lastSeq: number;
  sections: Record<string, string[]>;
  done: boolean;
  fetchUrl: string | null;
};

export function reduceStreamEvent(
  state: StreamReducerState | undefined,
  frame: SseFrame,
): StreamReducerState {
  const current = state ?? { lastSeq: 0, sections: {}, done: false, fetchUrl: null };
  const seq = Number(frame.data.seq ?? 0);
  if (seq <= current.lastSeq) {
    return current;
  }
  if (frame.event === "feedback_section_preview") {
    const section = String(frame.data.section);
    return {
      ...current,
      lastSeq: seq,
      sections: { ...current.sections, [section]: frame.data.items as string[] },
    };
  }
  if (frame.event === "done") {
    const result = frame.data.result as { fetch_url?: string } | undefined;
    return { ...current, lastSeq: seq, done: true, fetchUrl: result?.fetch_url ?? null };
  }
  return { ...current, lastSeq: seq };
}
```

export async function streamEssayFeedback(
  studentId: string,
  payload: EssayFeedbackStreamPayload,
  onFrame: (frame: EssayFeedbackStreamFrame) => void,
  options: { signal?: AbortSignal } = {},
): Promise<void> {
  const response = await fetch(apiUrl(`/api/students/${studentId}/essays/stream-feedback`), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
    signal: options.signal,
  });
  if (!response.ok || !response.body) {
    throw new Error("stream request failed");
  }
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const parts = buffer.split("\n\n");
    buffer = parts.pop() ?? "";
    for (const part of parts) {
      for (const frame of parseSseFrames(`${part}\n\n`)) {
        onFrame(frame);
      }
    }
  }
}
