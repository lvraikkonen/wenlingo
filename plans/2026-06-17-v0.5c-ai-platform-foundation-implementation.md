# V0.5c AI Platform Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Route every production AI task through one configurable, observable, fallback-capable, testable execution layer before V0.6 Writing Castle work begins.

**Architecture:** Add code-level routing registries for provider profiles, logical models, task configs, and pricing. Introduce a task-level `AITaskRunner`/`run_ai_task()` boundary that owns daily limits, provider execution, fallback, cost aggregation, and one-row-per-task logging while existing product wrappers keep payload, prompt, schema, validation, and deterministic fallback semantics. Migrate existing AI wrappers and routes onto the runner without adding new learning features or V0.6 prompt content.

**Tech Stack:** FastAPI, SQLModel, Alembic, Pydantic v2, httpx, pytest, Next.js App Router, React, TypeScript, Vitest, Testing Library, Playwright.

---

## Scope Guardrails

P0 V0.5c core is in scope:

- task -> logical model -> provider profile -> pricing registries
- OpenAI-compatible HTTP provider profiles
- task-level runner and fallback
- task-level daily limit semantics
- `LLMCallLog` observability fields and migration
- Admin usage aggregation
- docs and E2E coverage

P1/P2 items are documented at the end as V0.5c.1 follow-up candidates and should not be implemented while executing this plan unless Claus explicitly pulls them into scope.

```python
def configured_task_names() -> set[str]:
    return set(TASK_CONFIGS)


def resolve_task_config(task_name: str) -> TaskConfig:
    config = TASK_CONFIGS.get(task_name)
    if config is None:
        raise RoutingConfigError(f"Unknown AI task: {task_name}")
    if not config.enabled:
        raise RoutingConfigError(f"AI task is disabled: {task_name}")
    return config


def _model_with_override(settings: Settings, logical_model: LogicalModel) -> LogicalModel:
    override = getattr(settings, logical_model.model_env, "").strip() if logical_model.model_env else ""
    if not override:
        return logical_model
    pricing_key = f"{logical_model.provider_profile}:{override}"
    return LogicalModel(
        logical_model.logical_model_key,
        logical_model.provider_profile,
        override,
        pricing_key,
        logical_model.cost_tier,
        logical_model.model_env,
    )


def _pricing_for(settings: Settings, logical_model: LogicalModel) -> tuple[ModelPricing | None, str]:
    pricing = COST_REGISTRY.get(logical_model.pricing_key)
    if pricing is not None:
        return pricing, PricingStatus.CONFIGURED
    if settings.environment in {"staging", "production"}:
        raise RoutingConfigError(f"Missing pricing for routed model: {logical_model.pricing_key}")
    return None, PricingStatus.UNCONFIGURED


def resolve_task_route(settings: Settings, task_name: str, prompt_key: str | None = None) -> ResolvedTaskRoute:
    task = resolve_task_config(task_name)
    selected_prompt_key = prompt_key or task.default_prompt_key
    if selected_prompt_key not in task.allowed_prompt_keys:
        raise RoutingConfigError(f"Prompt key {selected_prompt_key} is not allowed for task {task_name}")
    primary_model = _model_with_override(settings, LOGICAL_MODELS[task.primary_model])
    fallback_model = _model_with_override(settings, LOGICAL_MODELS[task.fallback_model])
    primary_profile = PROVIDER_PROFILES[primary_model.provider_profile]
    fallback_profile = PROVIDER_PROFILES[fallback_model.provider_profile]
    primary_pricing, primary_status = _pricing_for(settings, primary_model)
    fallback_pricing, fallback_status = _pricing_for(settings, fallback_model)
    pricing_status = PricingStatus.CONFIGURED if primary_status == fallback_status == PricingStatus.CONFIGURED else PricingStatus.UNCONFIGURED
    return ResolvedTaskRoute(
        task=task,
        primary_model=primary_model,
        fallback_model=fallback_model,
        primary_profile=primary_profile,
        fallback_profile=fallback_profile,
        primary_pricing=primary_pricing,
        fallback_pricing=fallback_pricing,
        pricing_status=pricing_status,
        prompt_key=selected_prompt_key,
    )


def validate_ai_routing_startup(settings: Settings) -> None:
    for task in TASK_CONFIGS.values():
        if not task.enabled:
            continue
        route = resolve_task_route(settings, task.task_name, task.default_prompt_key)
        for profile in (route.primary_profile, route.fallback_profile):
            if profile.provider_type == "openai_compatible_http":
                base_url = getattr(settings, profile.base_url_env).strip()
                api_key = getattr(settings, profile.api_key_env).strip()
                if settings.environment in {"staging", "production"} and (not base_url or not api_key):
                    raise RoutingConfigError(f"Missing provider credentials for {profile.profile_name}")
```

```python
def provider_for_profile(
    *,
    settings: Settings,
    profile: ProviderProfile,
    logical_model: LogicalModel,
    timeout_seconds: int,
) -> LLMProvider:
    if profile.provider_type == "mock":
        return MockLLMProvider()
    if profile.provider_type == "openai_compatible_http":
        base_url = getattr(settings, profile.base_url_env).strip()
        api_key = getattr(settings, profile.api_key_env).strip()
        model_override = getattr(settings, logical_model.model_env).strip() if logical_model.model_env else ""
        model = model_override or logical_model.model
        missing = [
            name
            for name, value in {
                profile.base_url_env.upper(): base_url,
                profile.api_key_env.upper(): api_key,
                logical_model.model_env.upper() if logical_model.model_env else "MODEL": model,
            }.items()
            if not value
        ]
        if missing:
            raise HTTPException(
                status_code=503,
                detail=f"LLM provider profile {profile.profile_name} is missing configuration: {', '.join(missing)}",
            )
        return HttpJsonLLMProvider(
            api_key=api_key,
            model=model,
            base_url=base_url,
            timeout_seconds=timeout_seconds,
        )
    raise HTTPException(
        status_code=503,
        detail=f"Unsupported LLM provider profile type: {profile.provider_type}",
    )
```

```python
    role: str
    provider: str
    model: str
    status: str
    error_class: str
    latency_ms: int
    prompt_tokens: int = 0
    completion_tokens: int = 0
    total_tokens: int = 0
    estimated_cost: float = 0.0
    pricing_status: str = PricingStatus.UNAVAILABLE

    def to_summary(self) -> dict:
        return {
            "attempt_index": self.attempt_index,
            "role": self.role,
            "provider": self.provider,
            "model": self.model,
            "status": self.status,
            "error_class": self.error_class,
            "latency_ms": self.latency_ms,
            "prompt_tokens": self.prompt_tokens,
            "completion_tokens": self.completion_tokens,
            "total_tokens": self.total_tokens,
            "estimated_cost": self.estimated_cost,
            "pricing_status": self.pricing_status,
        }


def _usage_value(usage: dict[str, int] | None, key: str) -> int:
    if not usage:
        return 0
    return int(usage.get(key) or 0)


def _attempt_cost(prompt_tokens: int, completion_tokens: int, input_cost: float, output_cost: float) -> float:
    return (prompt_tokens / 1000 * input_cost) + (completion_tokens / 1000 * output_cost)
```

```python
async def _run_attempt(
    *,
    provider: LLMProvider,
    role: str,
    attempt_index: int,
    task_name: str,
    payload: dict,
    output_schema: type[T],
    validate_output: Callable[[T], None] | None,
    input_cost_per_1k: float,
    output_cost_per_1k: float,
    pricing_status: str,
) -> tuple[T | None, AttemptRecord, str | None, str]:
    started = perf_counter()
    provider_name = getattr(provider, "provider_name", provider.__class__.__name__)
    model_name = getattr(provider, "model_name", "unknown")
    try:
        response: LLMProviderResponse = await provider.complete_json(task_name, payload)
        latency_ms = int((perf_counter() - started) * 1000)
        provider_name = response.provider
        model_name = response.model
        prompt_tokens = _usage_value(response.usage, "prompt_tokens")
        completion_tokens = _usage_value(response.usage, "completion_tokens")
        total_tokens = _usage_value(response.usage, "total_tokens") or prompt_tokens + completion_tokens
        estimated_cost = _attempt_cost(prompt_tokens, completion_tokens, input_cost_per_1k, output_cost_per_1k)
        try:
            output = output_schema.model_validate(response.parsed_json)
        except ValidationError as exc:
            return (
                None,
                AttemptRecord(
                    attempt_index,
                    role,
                    provider_name,
                    model_name,
                    TaskFallbackReason.SCHEMA_VALIDATION_FAILED,
                    TaskFallbackReason.SCHEMA_VALIDATION_FAILED,
                    latency_ms,
                    prompt_tokens,
                    completion_tokens,
                    total_tokens,
                    estimated_cost,
                    pricing_status,
                ),
                str(exc),
                TaskFallbackReason.SCHEMA_VALIDATION_FAILED,
            )
        if validate_output is not None:
            try:
                validate_output(output)
            except Exception as exc:
                return (
                    None,
                    AttemptRecord(
                        attempt_index,
                        role,
                        provider_name,
                        model_name,
                        TaskFallbackReason.TASK_VALIDATION_FAILED,
                        TaskFallbackReason.TASK_VALIDATION_FAILED,
                        latency_ms,
                        prompt_tokens,
                        completion_tokens,
                        total_tokens,
                        estimated_cost,
                        pricing_status,
                    ),
                    str(exc),
                    TaskFallbackReason.TASK_VALIDATION_FAILED,
                )
        return (
            output,
            AttemptRecord(
                attempt_index,
                role,
                provider_name,
                model_name,
                "success",
                "",
                latency_ms,
                prompt_tokens,
                completion_tokens,
                total_tokens,
                estimated_cost,
                pricing_status,
            ),
            None,
            "",
        )
    except TimeoutError as exc:
        latency_ms = int((perf_counter() - started) * 1000)
        return None, AttemptRecord(attempt_index, role, provider_name, model_name, TaskFallbackReason.TIMEOUT, TaskFallbackReason.TIMEOUT, latency_ms), str(exc), TaskFallbackReason.TIMEOUT
    except Exception as exc:
        latency_ms = int((perf_counter() - started) * 1000)
        message = str(exc).lower()
        reason = TaskFallbackReason.RATE_LIMIT if "rate" in message and "limit" in message else TaskFallbackReason.API_ERROR
        return None, AttemptRecord(attempt_index, role, provider_name, model_name, reason, reason, latency_ms), str(exc), reason
```

```python
def _sum_attempts(attempts: list[AttemptRecord]) -> tuple[int, int, int, float, int]:
    prompt_tokens = sum(attempt.prompt_tokens for attempt in attempts)
    completion_tokens = sum(attempt.completion_tokens for attempt in attempts)
    total_tokens = sum(attempt.total_tokens for attempt in attempts)
    estimated_cost = sum(attempt.estimated_cost for attempt in attempts)
    latency_ms = sum(attempt.latency_ms for attempt in attempts)
    return prompt_tokens, completion_tokens, total_tokens, estimated_cost, latency_ms


def _record_log(
    *,
    session: Session | None,
    student_id: str | None,
    task_type: TaskType,
    task_name: str,
    prompt_key: str,
    prompt_version: str,
    input_summary: str,
    output: BaseModel,
    attempts: list[AttemptRecord],
    status: str,
    fallback_reason: str,
    pricing_status: str,
) -> LLMCallLog | None:
    if session is None:
        return None
    prompt_tokens, completion_tokens, total_tokens, estimated_cost, latency_ms = _sum_attempts(attempts)
    resolved_attempt = next((attempt for attempt in reversed(attempts) if attempt.status == "success"), None)
    primary_attempt = next((attempt for attempt in attempts if attempt.role == "primary"), None)
    fallback_attempt = next((attempt for attempt in attempts if attempt.role == "fallback"), None)
    provider = resolved_attempt.provider if resolved_attempt else "local_fallback"
    model = resolved_attempt.model if resolved_attempt else "local_fallback"
    log = LLMCallLog(
        student_id=student_id,
        task_type=task_type,
        task_name=task_name,
        prompt_key=prompt_key,
        provider=provider,
        model=model,
        resolved_provider=provider,
        resolved_model=model,
        primary_provider=primary_attempt.provider if primary_attempt else "",
        primary_model=primary_attempt.model if primary_attempt else "",
        fallback_provider=fallback_attempt.provider if fallback_attempt else "",
        fallback_model=fallback_attempt.model if fallback_attempt else "",
        fallback_reason=fallback_reason,
        attempt_count=len(attempts),
        final_status=status,
        pricing_status=pricing_status,
        attempt_summaries=[attempt.to_summary() for attempt in attempts],
        prompt_version=prompt_version,
        input_summary=input_summary,
        raw_response="",
        output_json=output.model_dump(),
        validation_ok=status in {TaskFinalStatus.PRIMARY_SUCCESS, TaskFinalStatus.FALLBACK_SUCCESS},
        error_message="" if status in {TaskFinalStatus.PRIMARY_SUCCESS, TaskFinalStatus.FALLBACK_SUCCESS} else fallback_reason,
        retry_count=max(len(attempts) - 1, 0),
        prompt_tokens=prompt_tokens,
        completion_tokens=completion_tokens,
        total_tokens=total_tokens,
        estimated_cost=estimated_cost,
        latency_ms=latency_ms,
    )
    session.add(log)
    session.flush()
    return log


async def run_ai_task(
    *,
    settings: Settings,
    session: Session | None,
    task_type: TaskType,
    task_name: str,
    student_id: str | None,
    payload: dict,
    output_schema: type[T],
    prompt_key: str,
    input_summary: str,
    deterministic_fallback_factory: Callable[[FailureContext], T] | None,
    validate_output: Callable[[T], None] | None = None,
    primary_provider: LLMProvider,
    fallback_provider: LLMProvider,
    daily_limit: int | None = None,
    prompt_version: str = "",
) -> AITaskResult[T]:
    route = resolve_task_route(settings, task_name, prompt_key)
    effective_daily_limit = daily_limit if daily_limit is not None else route.task.daily_limit
    if (
        session is not None
        and student_id is not None
        and settings.llm_daily_limit_enabled
        and llm_daily_limit_reached(
            session=session,
            student_id=student_id,
            task_name=task_name,
            limit=effective_daily_limit,
            timezone_name=settings.llm_daily_limit_timezone,
        )
    ):
        if deterministic_fallback_factory is None:
            raise RuntimeError(f"daily limit reached for {task_name}")
        output = deterministic_fallback_factory(
            FailureContext(task_name, TaskFinalStatus.DAILY_LIMIT_REACHED, ("daily limit reached",))
        )
        log = _record_log(
            session=session,
            student_id=student_id,
            task_type=task_type,
            task_name=task_name,
            prompt_key=prompt_key,
            prompt_version=prompt_version or prompt_key,
            input_summary=input_summary,
            output=output,
            attempts=[],
            status=TaskFinalStatus.DAILY_LIMIT_REACHED,
            fallback_reason="",
            pricing_status=route.pricing_status,
        )
        return AITaskResult(output=output, log=log, status=TaskFinalStatus.DAILY_LIMIT_REACHED)

    attempts: list[AttemptRecord] = []
    errors: list[str] = []
    primary_input_cost = route.primary_pricing.input_cost_per_1k_tokens if route.primary_pricing else 0.0
    primary_output_cost = route.primary_pricing.output_cost_per_1k_tokens if route.primary_pricing else 0.0
    output, attempt, error, reason = await _run_attempt(
        provider=primary_provider,
        role="primary",
        attempt_index=1,
        task_name=task_name,
        payload=payload,
        output_schema=output_schema,
        validate_output=validate_output,
        input_cost_per_1k=primary_input_cost,
        output_cost_per_1k=primary_output_cost,
        pricing_status=route.pricing_status,
    )
    attempts.append(attempt)
    if output is not None:
        log = _record_log(
            session=session,
            student_id=student_id,
            task_type=task_type,
            task_name=task_name,
            prompt_key=prompt_key,
            prompt_version=prompt_version or prompt_key,
            input_summary=input_summary,
            output=output,
            attempts=attempts,
            status=TaskFinalStatus.PRIMARY_SUCCESS,
            fallback_reason="",
            pricing_status=route.pricing_status,
        )
        return AITaskResult(output=output, log=log, status=TaskFinalStatus.PRIMARY_SUCCESS)
    errors.append(error or reason)

    fallback_input_cost = route.fallback_pricing.input_cost_per_1k_tokens if route.fallback_pricing else 0.0
    fallback_output_cost = route.fallback_pricing.output_cost_per_1k_tokens if route.fallback_pricing else 0.0
    fallback_output, fallback_attempt, fallback_error, fallback_reason = await _run_attempt(
        provider=fallback_provider,
        role="fallback",
        attempt_index=2,
        task_name=task_name,
        payload=payload,
        output_schema=output_schema,
        validate_output=validate_output,
        input_cost_per_1k=fallback_input_cost,
        output_cost_per_1k=fallback_output_cost,
        pricing_status=route.pricing_status,
    )
    attempts.append(fallback_attempt)
    if fallback_output is not None:
        log = _record_log(
            session=session,
            student_id=student_id,
            task_type=task_type,
            task_name=task_name,
            prompt_key=prompt_key,
            prompt_version=prompt_version or prompt_key,
            input_summary=input_summary,
            output=fallback_output,
            attempts=attempts,
            status=TaskFinalStatus.FALLBACK_SUCCESS,
            fallback_reason=reason,
            pricing_status=route.pricing_status,
        )
        return AITaskResult(output=fallback_output, log=log, status=TaskFinalStatus.FALLBACK_SUCCESS)
    errors.append(fallback_error or fallback_reason)

    if deterministic_fallback_factory is not None:
        output = deterministic_fallback_factory(
            FailureContext(task_name, fallback_reason or reason or TaskFallbackReason.UNKNOWN_ERROR, tuple(errors))
        )
        log = _record_log(
            session=session,
            student_id=student_id,
            task_type=task_type,
            task_name=task_name,
            prompt_key=prompt_key,
            prompt_version=prompt_version or prompt_key,
            input_summary=input_summary,
            output=output,
            attempts=attempts,
            status=TaskFinalStatus.DETERMINISTIC_FALLBACK_USED,
            fallback_reason=fallback_reason or reason or TaskFallbackReason.UNKNOWN_ERROR,
            pricing_status=route.pricing_status,
        )
        return AITaskResult(output=output, log=log, status=TaskFinalStatus.DETERMINISTIC_FALLBACK_USED)

    raise RuntimeError("; ".join(errors))
```

```markdown
V0.5c routes AI tasks through code-level task and model registries. Environment variables provide credentials and deployment-specific model overrides only.
```
