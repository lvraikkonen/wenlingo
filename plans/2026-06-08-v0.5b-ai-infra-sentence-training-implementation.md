# V0.5b AI Infrastructure + Sentence Training Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the V0.5b AI infrastructure preflight and ship a thin vertical slice of AI-generated sentence challenges inside the existing sentence workshop.

**Architecture:** Keep `SentenceTraining` as the canonical sentence-practice table, extending it from completed-only rows to generated-then-completed challenge rows. Add a code-level prompt registry and task-specific guardrails around the existing LLM provider/task runner instead of creating an Admin prompt system or a new training engine. Frontend changes reuse the current child sentence page, Admin Alpha Lite page, and parent summary surfaces.

**Tech Stack:** FastAPI, SQLModel, Alembic, Pydantic v2, httpx, pytest, Next.js App Router, React, TypeScript, Vitest, Testing Library.

---

## Scope Guardrails

V0.5b must stay inside these boundaries:

- Do not build Writing Castle or any pre-writing flow.
- Do not build an Admin prompt editor.
- Do not build Prompt A/B testing.
- Do not build Full Admin v2.
- Do not build a complete sentence curriculum.
- Do not seed or curate a large question bank.
- Do not change the V0.5a auth/session foundation.

The implementation should only add the infrastructure and sentence-challenge vertical slice described in `specs/2026-06-08-v0.5b-ai-infra-sentence-training-design.md`.

```python
from dataclasses import dataclass
from typing import Callable


PayloadBuilder = Callable[[], dict]


@dataclass(frozen=True)
class PromptSpec:
    prompt_key: str
    version: str
    system_prompt_key: str
    system_prompt: str
    response_contract: str
    build_payload: PayloadBuilder | None = None


_PROMPTS: dict[str, PromptSpec] = {}


def register_prompt(prompt: PromptSpec) -> PromptSpec:
    if prompt.prompt_key in _PROMPTS:
        raise ValueError(f"duplicate prompt key: {prompt.prompt_key}")
    _PROMPTS[prompt.prompt_key] = prompt
    return prompt


def get_prompt(prompt_key: str) -> PromptSpec:
    return _PROMPTS[prompt_key]


def registered_prompts() -> dict[str, PromptSpec]:
    return dict(_PROMPTS)
```

```python
from datetime import UTC, datetime, time
from zoneinfo import ZoneInfo

from sqlmodel import Session, func, select

from app.domain.models import LLMCallLog


REAL_PROVIDER_NAMES = {"http"}


def local_day_start_utc(now: datetime, timezone_name: str) -> datetime:
    zone = ZoneInfo(timezone_name)
    local_now = now.astimezone(zone)
    local_start = datetime.combine(local_now.date(), time.min, tzinfo=zone)
    return local_start.astimezone(UTC)


def is_real_provider(provider_name: str) -> bool:
    return provider_name in REAL_PROVIDER_NAMES


def llm_daily_limit_reached(
    *,
    session: Session,
    student_id: str,
    task_name: str,
    provider_name: str,
    limit: int,
    timezone_name: str,
    now: datetime | None = None,
) -> bool:
    if limit <= 0 or not is_real_provider(provider_name):
        return False
    start = local_day_start_utc(now or datetime.now(UTC), timezone_name)
    count = session.exec(
        select(func.count(LLMCallLog.id)).where(
            LLMCallLog.student_id == student_id,
            LLMCallLog.task_name == task_name,
            LLMCallLog.provider == provider_name,
            LLMCallLog.created_at >= start,
        )
    ).one()
    return int(count or 0) >= limit
```

def alpha_admin_ai_usage(session: Session = Depends(get_db_session)):
    logs = session.exec(select(LLMCallLog)).all()
    events = session.exec(
        select(ProductEvent).where(ProductEvent.event_type == "ai_daily_limit_reached")
    ).all()

    limit_hits: dict[tuple[str, str], int] = {}
    for event in events:
        task_type = str(event.payload.get("task_type") or "unknown")
        date_key = event.created_at.date().isoformat()
        limit_hits[(date_key, task_type)] = limit_hits.get((date_key, task_type), 0) + 1

    grouped: dict[tuple[str, str, str], dict[str, Any]] = {}
    for log in logs:
        date_key = log.created_at.date().isoformat()
        key = (date_key, log.task_name, log.model)
        row = grouped.setdefault(
            key,
            {
                "date": date_key,
                "task_type": log.task_name,
                "model": log.model,
                "call_count": 0,
                "prompt_tokens": 0,
                "completion_tokens": 0,
                "total_tokens": 0,
                "estimated_cost": 0.0,
                "failure_count": 0,
                "daily_limit_hit_count": 0,
            },
        )
        row["call_count"] += 1
        row["prompt_tokens"] += log.prompt_tokens
        row["completion_tokens"] += log.completion_tokens
        row["total_tokens"] += log.total_tokens
        row["estimated_cost"] += log.estimated_cost
        if not log.validation_ok or log.error_message:
            row["failure_count"] += 1

    for row in grouped.values():
        row["estimated_cost"] = round(row["estimated_cost"], 6)
        row["daily_limit_hit_count"] = limit_hits.get((row["date"], row["task_type"]), 0)
