# V0.4.1b Alpha Feedback & Observation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the approved V0.4.1b Alpha feedback and observation loop: one-time invite codes, privacy-safe product events, child reactions, parent summary feedback, and Admin Alpha Lite.

**Architecture:** Keep the Alpha surface additive and narrow. Add four SQLModel tables plus one migration, extend the existing Alpha route for invite/events/parent feedback, add a focused reaction route for learning feedback targets, and add a small admin route protected by `ALPHA_ADMIN_TOKEN`. On the web side, preserve the V0.4.1a Alpha flow, add `wenlingo_alpha_session_id`, upgrade `/alpha/start` to invitation onboarding, add reusable feedback controls, and build a read-only admin page with overview plus a simple family timeline.

**Tech Stack:** FastAPI, SQLModel, Alembic-style migration files under `apps/api/app/db/migrations/versions`, Pydantic, pytest, Next.js App Router, React client components, Vitest, Testing Library, `localStorage`, `sessionStorage`, Railway PostgreSQL.

## Non-Negotiables

- Do not add full authentication.
- Do not add public signup.
- Do not turn `ProductEvent` into a full analytics system.
- Do not store child writing text, upgraded sentence text, essay text, or AI feedback body in `ProductEvent.payload`.
- Do not show child writing text or AI feedback body in `/admin/alpha`.
- Do not block the learning flow if product event logging fails.
- Do not block the child from continuing if reaction saving fails.
- Do not create a formal `Report` entity just for parent summary feedback.
- Do not rewrite existing assessment, sentence, or essay flows except for small event/reaction integration.
- Demo flow must remain intact.
- Existing V0.4.1a Alpha families must continue without re-entering invite codes.

```python
def hash_invite_code(code: str) -> str:
    return sha256(code.strip().upper().encode("utf-8")).hexdigest()


def sanitize_event_payload(payload: dict[str, object] | None) -> dict[str, object]:
    if not payload:
        return {}
    return {key: value for key, value in payload.items() if key in SAFE_PAYLOAD_KEYS}


def record_product_event(
    session: Session,
    event_type: str,
    *,
    parent_id: str | None = None,
    student_id: str | None = None,
    invite_code_id: str | None = None,
    alpha_session_id: str = "",
    payload: dict[str, object] | None = None,
) -> None:
    if event_type not in ALLOWED_EVENT_TYPES:
        raise ValueError("unknown product event type")
    session.add(
        ProductEvent(
            event_type=event_type,
            parent_id=parent_id,
            student_id=student_id,
            invite_code_id=invite_code_id,
            alpha_session_id=alpha_session_id or "",
            payload=sanitize_event_payload(payload),
        )
    )
```

```python
@router.post("/invites/validate")
def validate_alpha_invite(
    request: AlphaInviteValidate,
    session: Session = Depends(get_db_session),
):
    invite = session.exec(
        select(AlphaInviteCode).where(
            AlphaInviteCode.code_hash == hash_invite_code(request.code)
        )
    ).first()
    if not invite or invite.status != "issued":
        record_product_event(
            session,
            "invite_code_rejected",
            alpha_session_id=request.alpha_session_id,
            payload={"status": "rejected", "error_category": "not_available"},
        )
        session.commit()
        raise HTTPException(status_code=400, detail="invite code is not available")
    record_product_event(
        session,
        "invite_code_validated",
        invite_code_id=invite.id,
        alpha_session_id=request.alpha_session_id,
        payload={"status": "validated"},
    )
    session.commit()
    return {"valid": True, "invite_id": invite.id, "label": invite.label}


@router.post("/events", status_code=201)
def create_product_event(
    request: ProductEventCreate,
    session: Session = Depends(get_db_session),
):
    try:
        record_product_event(
            session,
            request.event_type,
            parent_id=request.parent_id,
            student_id=request.student_id,
            invite_code_id=request.invite_code_id,
            alpha_session_id=request.alpha_session_id,
            payload=request.payload,
        )
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    session.commit()
    return {"ok": True}
```

```typescript
export const ALPHA_SESSION_STORAGE_KEY = "wenlingo_alpha_session_id";

function randomSessionId() {
  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    return crypto.randomUUID();
  }
  return `alpha-session-${Date.now()}-${Math.random().toString(36).slice(2)}`;
}

export function getStoredAlphaSessionId(): string {
  if (typeof window === "undefined") {
    return "";
  }
  try {
    const existing = window.localStorage.getItem(ALPHA_SESSION_STORAGE_KEY);
    if (existing) {
      return existing;
    }
    const next = randomSessionId();
    window.localStorage.setItem(ALPHA_SESSION_STORAGE_KEY, next);
    return next;
  } catch {
    return "";
  }
}
```
