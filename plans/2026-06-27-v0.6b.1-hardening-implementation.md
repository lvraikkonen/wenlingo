# V0.6b.1 Hardening Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the approved V0.6b.1 hardening spec: expository fact-boundary prompt contracts, person-portrait child-view material-question contracts, and safe Dev QA child-profile cleanup.

**Architecture:** Keep Prompt Registry as the only prompt source and change only the two task-level Writing Castle prompt contracts that V0.6b.1 owns. Add a focused backend cleanup service for QA child profiles, with preview and execute sharing the same matcher, execute guarded by environment and confirmation, and all deletes scoped to matched child ids or child-owned artifacts. Expose cleanup through an ops CLI only; do not add Admin UI or API surface in this release.

**Tech Stack:** FastAPI/SQLModel backend, Pydantic prompt contracts, pytest, existing ops CLI patterns, existing Prompt Registry and Writing Castle services.

from dataclasses import dataclass, field
import os
from typing import Any

from sqlmodel import Session, select

from app.core.config import Settings, get_settings
from app.domain.models import (
    AbilityHistory,
    AbilityProfile,
    Assessment,
    DailyTaskLimitCounter,
    Essay,
    EssayVersion,
    FeedbackReaction,
    GameEvent,
    LLMCallLog,
    ParentFeedback,
    ProductEvent,
    ReadingSession,
    Report,
    SentenceTraining,
    StudentProfile,
)

DELETE_QA_CHILD_PROFILES_CONFIRMATION = "DELETE QA CHILD PROFILES"
MAX_QA_CHILD_PROFILE_DELETE = 30
QA_CHILD_EXACT_NAMES = {"QA v0.6b"}
QA_CHILD_PREFIXES = ("QA06b-",)
DEV_TEST_ENVIRONMENTS = {"development", "dev", "test"}


class QAChildProfileCleanupError(RuntimeError):
    pass


@dataclass(frozen=True)
class CleanupEnvironment:
    environment: str
    railway_environment_name: str
    execute_allowed: bool


@dataclass
class QAChildProfileCleanupChild:
    student_id: str
    child_name: str
    parent_id: str
    essay_ids: list[str] = field(default_factory=list)
    record_counts: dict[str, int] = field(default_factory=dict)


@dataclass
class QAChildProfileCleanupResult:
    mode: str
    detected_environment: CleanupEnvironment
    children: list[QAChildProfileCleanupChild]

    @property
    def matched_count(self) -> int:
        return len(self.children)

    @property
    def deleted_count(self) -> int:
        return len(self.children) if self.mode == "execute" else 0


def is_v06b_qa_child_name(name: str) -> bool:
    normalized = " ".join(str(name or "").split())
    return normalized in QA_CHILD_EXACT_NAMES or normalized.startswith(QA_CHILD_PREFIXES)


def detect_cleanup_environment(
    settings: Settings | None = None,
    *,
    railway_environment_name: str | None = None,
) -> CleanupEnvironment:
    effective_settings = settings or get_settings()
    environment = str(effective_settings.environment or "").strip().lower()
    railway_name = (
        os.environ.get("RAILWAY_ENVIRONMENT_NAME", "")
        if railway_environment_name is None
        else railway_environment_name
    )
    execute_allowed = environment in DEV_TEST_ENVIRONMENTS or "dev" in railway_name.lower()
    return CleanupEnvironment(
        environment=environment,
        railway_environment_name=railway_name,
        execute_allowed=execute_allowed,
    )


def _rows_where_in(session: Session, model: Any, field: Any, values: set[str]) -> list[Any]:
    if not values:
        return []
    return list(session.exec(select(model).where(field.in_(values))).all())


def _count_where_in(session: Session, model: Any, field: Any, values: set[str]) -> int:
    return len(_rows_where_in(session, model, field, values))


def _delete_where_in(session: Session, model: Any, field: Any, values: set[str]) -> int:
    rows = _rows_where_in(session, model, field, values)
    for row in rows:
        session.delete(row)
    return len(rows)


def _matched_children(session: Session) -> list[StudentProfile]:
    children = session.exec(select(StudentProfile)).all()
    matched = [child for child in children if is_v06b_qa_child_name(child.name)]
    return sorted(matched, key=lambda child: (child.created_at, child.id))


def _preview_child(session: Session, child: StudentProfile) -> QAChildProfileCleanupChild:
    student_ids = {child.id}
    essays = _rows_where_in(session, Essay, Essay.student_id, student_ids)
    essay_ids = {essay.id for essay in essays}
    counts = {
        "Assessment": _count_where_in(session, Assessment, Assessment.student_id, student_ids),
        "EssayVersion": _count_where_in(session, EssayVersion, EssayVersion.essay_id, essay_ids),
        "Essay": len(essays),
        "SentenceTraining": _count_where_in(session, SentenceTraining, SentenceTraining.student_id, student_ids),
        "ReadingSession": _count_where_in(session, ReadingSession, ReadingSession.student_id, student_ids),
        "GameEvent": _count_where_in(session, GameEvent, GameEvent.student_id, student_ids),
        "Report": _count_where_in(session, Report, Report.student_id, student_ids),
        "AbilityHistory": _count_where_in(session, AbilityHistory, AbilityHistory.student_id, student_ids),
        "AbilityProfile": _count_where_in(session, AbilityProfile, AbilityProfile.student_id, student_ids),
        "DailyTaskLimitCounter": _count_where_in(session, DailyTaskLimitCounter, DailyTaskLimitCounter.student_id, student_ids),
        "LLMCallLog": _count_where_in(session, LLMCallLog, LLMCallLog.student_id, student_ids),
        "FeedbackReaction": _count_where_in(session, FeedbackReaction, FeedbackReaction.student_id, student_ids),
        "ParentFeedback": _count_where_in(session, ParentFeedback, ParentFeedback.student_id, student_ids),
        "ProductEvent": _count_where_in(session, ProductEvent, ProductEvent.student_id, student_ids),
        "StudentProfile": 1,
    }
    return QAChildProfileCleanupChild(
        student_id=child.id,
        child_name=child.name,
        parent_id=child.parent_id,
        essay_ids=sorted(essay_ids),
        record_counts=counts,
    )


def preview_qa_child_profile_cleanup(
    session: Session,
    *,
    settings: Settings | None = None,
    railway_environment_name: str | None = None,
) -> QAChildProfileCleanupResult:
    detected_environment = detect_cleanup_environment(
        settings,
        railway_environment_name=railway_environment_name,
    )
    return QAChildProfileCleanupResult(
        mode="dry-run",
        detected_environment=detected_environment,
        children=[_preview_child(session, child) for child in _matched_children(session)],
    )


def _delete_child_rows(
    session: Session,
    row: QAChildProfileCleanupChild,
) -> dict[str, int]:
    student_ids = {row.student_id}
    essay_ids = set(row.essay_ids)
    counts: dict[str, int] = {}
    counts["Assessment"] = _delete_where_in(session, Assessment, Assessment.student_id, student_ids)
    counts["EssayVersion"] = _delete_where_in(session, EssayVersion, EssayVersion.essay_id, essay_ids)
    counts["Essay"] = _delete_where_in(session, Essay, Essay.student_id, student_ids)
    counts["SentenceTraining"] = _delete_where_in(session, SentenceTraining, SentenceTraining.student_id, student_ids)
    counts["ReadingSession"] = _delete_where_in(session, ReadingSession, ReadingSession.student_id, student_ids)
    counts["GameEvent"] = _delete_where_in(session, GameEvent, GameEvent.student_id, student_ids)
    counts["Report"] = _delete_where_in(session, Report, Report.student_id, student_ids)
    counts["AbilityHistory"] = _delete_where_in(session, AbilityHistory, AbilityHistory.student_id, student_ids)
    counts["AbilityProfile"] = _delete_where_in(session, AbilityProfile, AbilityProfile.student_id, student_ids)
    counts["DailyTaskLimitCounter"] = _delete_where_in(session, DailyTaskLimitCounter, DailyTaskLimitCounter.student_id, student_ids)
    counts["LLMCallLog"] = _delete_where_in(session, LLMCallLog, LLMCallLog.student_id, student_ids)
    counts["FeedbackReaction"] = _delete_where_in(session, FeedbackReaction, FeedbackReaction.student_id, student_ids)
    counts["ParentFeedback"] = _delete_where_in(session, ParentFeedback, ParentFeedback.student_id, student_ids)
    counts["ProductEvent"] = _delete_where_in(session, ProductEvent, ProductEvent.student_id, student_ids)
    counts["StudentProfile"] = _delete_where_in(session, StudentProfile, StudentProfile.id, student_ids)
    return counts


def cleanup_qa_child_profiles(
    session: Session,
    *,
    confirm: str,
    settings: Settings | None = None,
    railway_environment_name: str | None = None,
    max_children: int = MAX_QA_CHILD_PROFILE_DELETE,
) -> QAChildProfileCleanupResult:
    detected_environment = detect_cleanup_environment(
        settings,
        railway_environment_name=railway_environment_name,
    )
    if not detected_environment.execute_allowed:
        raise QAChildProfileCleanupError(
            "refusing to execute QA child cleanup outside Dev/Test environment"
        )
    if confirm != DELETE_QA_CHILD_PROFILES_CONFIRMATION:
        raise QAChildProfileCleanupError("confirmation text is required")

    preview = preview_qa_child_profile_cleanup(
        session,
        settings=settings,
        railway_environment_name=railway_environment_name,
    )
    if preview.matched_count > max_children:
        raise QAChildProfileCleanupError(
            f"matched child count exceeds limit: {preview.matched_count} > {max_children}"
        )

    try:
        for row in preview.children:
            row.record_counts = _delete_child_rows(session, row)
        session.commit()
    except Exception:
        session.rollback()
        raise

    return QAChildProfileCleanupResult(
        mode="execute",
        detected_environment=detected_environment,
        children=preview.children,
    )
```

import argparse

from sqlmodel import Session

from app.db.session import engine
from app.services.qa_child_profile_cleanup import (
    DELETE_QA_CHILD_PROFILES_CONFIRMATION,
    QAChildProfileCleanupError,
    cleanup_qa_child_profiles,
    preview_qa_child_profile_cleanup,
)


def _parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Clean up V0.6b Dev QA child profiles.")
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument(
        "--dry-run",
        action="store_false",
        dest="execute",
        help="Preview matching QA child profiles without deleting them. This is the default.",
    )
    mode.add_argument(
        "--execute",
        action="store_true",
        dest="execute",
        help="Delete matching QA child profiles after confirmation.",
    )
    parser.set_defaults(execute=False)
    parser.add_argument(
        "--confirm",
        default="",
        help=f'Required with --execute. Use exactly "{DELETE_QA_CHILD_PROFILES_CONFIRMATION}".',
    )
    return parser.parse_args(argv)


def _print_result(result) -> None:
    env = result.detected_environment
    print(
        "mode={mode} environment={environment} railway_environment={railway} "
        "execute_allowed={execute_allowed} matched_count={matched_count} deleted_count={deleted_count}".format(
            mode=result.mode,
            environment=env.environment,
            railway=env.railway_environment_name or "-",
            execute_allowed=env.execute_allowed,
            matched_count=result.matched_count,
            deleted_count=result.deleted_count,
        )
    )
    for child in result.children:
        print(
            "student_id={student_id} child_name={child_name} parent_id={parent_id} "
            "essay_ids={essay_ids} record_counts={record_counts}".format(
                student_id=child.student_id,
                child_name=child.child_name,
                parent_id=child.parent_id,
                essay_ids=",".join(child.essay_ids) if child.essay_ids else "-",
                record_counts=child.record_counts,
            )
        )


def main(argv: list[str] | None = None) -> int:
    args = _parse_args(argv)
    with Session(engine) as session:
        try:
            if args.execute:
                result = cleanup_qa_child_profiles(session, confirm=args.confirm)
            else:
                result = preview_qa_child_profile_cleanup(session)
        except QAChildProfileCleanupError as exc:
            raise SystemExit(str(exc)) from exc

    _print_result(result)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
