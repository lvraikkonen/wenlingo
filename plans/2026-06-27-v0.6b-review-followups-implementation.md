```python
def _valid_reading_material_refs(answers: list[dict]) -> set[str]:
    refs: set[str] = set()
    for answer in answers:
        for ref in answer.get("source_refs", []) if isinstance(answer.get("source_refs"), list) else []:
            if not isinstance(ref, dict):
                continue
            if ref.get("source_type") != "reading_material":
                continue
            ref_id = _payload_source_id(ref.get("reading_material_ref"))
            if ref_id is not None:
                refs.add(ref_id)
    return refs


def _validate_material_card_source_ref_provenance(
    output: MaterialCardsResult,
    *,
    valid_answer_ids: set[str],
    valid_reading_material_refs: set[str],
) -> None:
    unknown_answer_ids = set()
    unknown_reading_material_refs = set()
    for card in output.cards:
        for ref in card.source_refs:
            source_type = str(ref.get("source_type") or "").strip()
            if source_type == "child_confirmed":
                raise LLMTaskValidationError("model output cannot create child_confirmed source_refs")
            answer_id = _payload_source_id(ref.get("answer_id"))
            if answer_id is not None and answer_id not in valid_answer_ids:
                unknown_answer_ids.add(answer_id)
            if source_type == "reading_material":
                material_ref = _payload_source_id(ref.get("reading_material_ref"))
                if material_ref is None or material_ref not in valid_reading_material_refs:
                    unknown_reading_material_refs.add(str(ref.get("reading_material_ref") or ""))
    if unknown_answer_ids:
        raise LLMTaskValidationError("unknown source_ref answer_ids: " + ", ".join(sorted(unknown_answer_ids)))
    if unknown_reading_material_refs:
        raise LLMTaskValidationError(
            "unknown reading_material_ref: " + ", ".join(sorted(unknown_reading_material_refs))
        )
```
